\section{Proof for Lemma \ref{lem:cconv'-splitting}}
\label{appendix:section:proof-splitting-lemma}

\begin{proof}
    \label{proof:cconv'-splitting}
    Let $\sigma_\sys \in \Traces_\sys$ and $\tau \in \Nat$.
    The proof for Lemma \ref{lem:cconv'-splitting} proceeds
    by structural induction over system traces
    $\sigma_\sys \in \Traces_\sys$.
    \begin{itemize}
        \item \textbf{Base Case} $\sigma_\sys = \emptytrace_\sys$
        \begin{align*}
            \cConv'(\tau, \sigma_\sys \cdot D)
            &=  \cConv'(\tau, \emptytrace_\sys \cdot D) \\
            &=  \cConv'(\tau, D) \\
            &=  \emptytrace \cdot \cConv'(\tau + 0, D) \\
            &\stackrel{\ref{def:num-ticks}}{=} 
                \emptytrace \cdot
                \cConv'(\tau + \numTicks(\emptytrace_\sys), D) \\
            &\stackrel{\ref{def:cconv'}}{=} 
                \cConv'(0, \emptytrace_\sys) \cdot
                \cConv'(\tau + \numTicks(\emptytrace_\sys), D) \\
            &= \cConv'(0, \sigma_\sys) \cdot
                \cConv'(\tau + \numTicks(\sigma_\sys), D)
        \end{align*}
        \item \textbf{Induction Hypothesis}

        For any true suffix $\sigma_\sys'$ of $\sigma_\sys$ 
        (i.e. $|\sigma_\sys'| < |\sigma_\sys|$)
        $$
        \cConv'(\tau, \sigma_\sys' \cdot D)
        = \cConv'(\tau, \sigma_\sys') \cdot
        \cConv(\tau + \numTicks(\sigma_\sys'), D).
        $$

        \item \textbf{Induction Step}

        We make a case distinction based on $\sigma_\sys$.
        Without loss of generality we can assume that there
        are no $\sskip$ events in $\sigma_\sys$.
        \begin{itemize}
            \item $\sigma_\sys = \tick^k$
            \begin{align*}
                \cConv'(\tau, \sigma_\sys \cdot D)
                &= \cConv'(\tau, \tick^k \cdot D) \\
                &\stackrel{\ref{def:cconv'}}{=}
                    (\tau + k, D) \\
                &= \emptytrace \cdot
                    (\tau + k, D) \\
                &\stackrel{\ref{def:cconv'}}{=}
                    \cConv(\tau, \emptytrace_\sys) \cdot
                    (\tau + k, D) \\
                &= \cConv(\tau, \tick^k) \cdot
                    (\tau + k, D) \\
                &= \cConv(\tau, \sigma_\sys) \cdot
                    (\tau + k, D) \\
                &\stackrel{\ref{def:cconv'}}{=}
                    \cConv(\tau, \sigma_\sys) \cdot
                    \cConv'(\tau+k, D) \\
                &= \cConv(\tau, \sigma_\sys) \cdot
                    \cConv'(\tau+ \numTicks(\sigma_\sys), D)
            \end{align*}
            \item $\sigma_\sys = \tick^k \cdot D' \cdot \sigma_\sys'$
            \begin{align*}
                \cConv'(\tau, \sigma_\sys \cdot D)
                &= \cConv'(\tau, \tick^k \cdot D' \cdot \sigma_\sys' \cdot D) \\
                &\stackrel{\ref{def:cconv'}}{=}
                    (\tau+k, D') \cdot
                    \cConv'(\tau+k, \sigma_\sys' \cdot D) \\
                &\stackrel{IH}{=}
                    (\tau+k, D') \cdot
                    \cConv'(\tau+k, \sigma_\sys') \cdot
                    \cConv'(\tau+k+\numTicks(\sigma_\sys'), D) \\
                &= (\tau+k, D') \cdot
                    \cConv'(\tau+k, \sigma_\sys') \cdot
                    \cConv'(\tau+\numTicks(\sigma_\sys), D) \\
                &\stackrel{\ref{def:cconv'}}{=}
                    \cConv'(\tau, \tick^k \cdot D' \cdot \sigma_\sys') \cdot
                    \cConv'(\tau+\numTicks(\sigma_\sys), D) \\
                &= \cConv'(\tau, \sigma_\sys) \cdot
                    \cConv'(\tau+\numTicks(\sigma_\sys), D)
            \end{align*}
        \end{itemize}
        
    \end{itemize}
\end{proof}

\section{Proof for Lemma \ref{lem:enf-es-trace-correspondance}}
\label{appendix:section:proof-enf-es-trace-correspondance}

\begin{proof}
    \label{proof:enforcement-event-system-traces-corresponding-enforced-trace}
    The proof for Lemma
    \ref{lem:enf-es-trace-correspondance}
    proceeds by structural induction over finite system traces
    $\sigma_\sys \in \EStraces_f(\EventSystem(\Enforcer), \emptytrace_\sys)$.
    Later we will make the argument why the property extends to inifinite
    system traces too.
    Without loss of generality we can assume that
    $\sigma_\sys \in \trim(\Traces_{\sys})$,
    because the conversion function $\cConv$ removes
    implicitly trims system traces.
    Let $\Enforcer = (\EnfStates, s_0, \mu)$ be an enforcer.
    \begin{itemize}
        \item
        \textbf{Base Case} $\sigma_\sys = \emptytrace_\sys$
        \begin{align*}
            \cConv(\sigma_\sys)
            = \cConv(\emptytrace_\sys) 
            \stackrel{\ref{def:cconv}}{=} \emptytrace
            = \Enforcer(\emptytrace)
        \end{align*}
        The final equality follows from Algorithm \ref{alg:loop}
        which defines, that enforcing the empty trace returns
        the empty trace.        
        \item \textbf{Induction Hypothesis}
        For all true prefixes $\sigma_\sys'$ of
        $\sigma_\sys \in \EStraces_f(\EventSystem(\Enforcer), \emptytrace_\sys)$
        (i.e. $|\sigma_\sys'| < |\sigma_\sys|$)
        there exists a trace $\sigma' \in \Traces$ such that
        $\cConv(\sigma_\sys') = \Enforcer(\sigma')$.

        \item \textbf{Induction Step}

        We make a case distinction based on the two possible transitions
        of enforcement event systems from Definition
        \ref{def:enf-transition-relation}.
        The update functions $\RFun$ and $\PFun$ are defined
        as in Definition \ref{def:induced-event-system}.
        \begin{itemize}
            \item
            $\sigma_\sys = \sigma_\sys' \cdot \RFun(\sigma_\sys', D)$
            \begin{align*}
                \cConv(\sigma_\sys)
                &= \cConv(\sigma_\sys' \cdot \RFun(\sigma_\sys', D)) \\
                &\stackrel{\ref{def:cconv}}{=}
                    \cConv'(0, \sigma_\sys' \cdot \RFun(\sigma_\sys', D)) \\
                &\stackrel{\ref{lem:cconv'-splitting}}{=}
                    \cConv'(0, \sigma_\sys') \cdot
                    \cConv'(\numTicks(\sigma_\sys'), \RFun(\sigma_\sys', D)) \\
                &\stackrel{\ref{def:cconv}}{=}
                    \cConv(\sigma_\sys') \cdot
                    \cConv'(\numTicks(\sigma_\sys'), \RFun(\sigma_\sys', D)) \\
                &\stackrel{IH}{=}
                    \Enforcer(\sigma') \cdot
                    \cConv'(\numTicks(\sigma_\sys'), \RFun(\sigma_\sys', D)) \\
                &\stackrel{\ref{def:cconv'}, \ref{def:reactive-update-function}}{=}
                    \Enforcer(\sigma') \cdot
                    (\numTicks(\sigma_\sys'), \RFun(\sigma_\sys', D)) \\
                &=
                    \Enforcer(\sigma' \cdot
                    (\numTicks(\sigma_\sys'), D))
            \end{align*}
            The final step follows from the definition of the
            reactive update function $\RFun$ by means of the enforcer
            update function $\mu$ and how $\mu$ is used by
            in lines 6 and 7 of Algorithm \ref{alg:loop}
            to issue an $\RCom$-command.

            \item $\sigma_\sys = \sigma_\sys' \cdot \PFun(\sigma_\sys') \cdot \tick$
            \begin{align*}
                \cConv(\sigma_\sys)
                &= \cConv(\sigma_\sys' \cdot \PFun(\sigma_\sys') \cdot \tick) \\
                &\stackrel{\ref{def:cconv'}}{=}
                    \cConv(\sigma_\sys' \cdot \PFun(\sigma_\sys')) \\
                &\stackrel{\ref{def:cconv}}{=}
                    \cConv'(0, \sigma_\sys' \cdot \PFun(\sigma_\sys')) \\
                &\stackrel{\ref{lem:cconv'-splitting}, \sskip}{=}
                    \cConv'(0, \sigma_\sys') \cdot
                    \cConv'(\numTicks(\sigma_\sys',
                    \PFun(\sigma_\sys'))) \\
                &\stackrel{\ref{def:cconv}}{=}
                    \cConv(\sigma_\sys') \cdot
                    \cConv'(\numTicks(\sigma_\sys',
                    \PFun(\sigma_\sys'))) \\
                &\stackrel{IH}{=}
                    \Enforcer(\sigma') \cdot
                    \cConv'(\numTicks(\sigma_\sys',
                    \PFun(\sigma_\sys'))) \\
                &= \begin{cases}
                    \Enforcer(\sigma')
                        &\text{if } \PFun(\sigma_\sys')  = \sskip \\
                    \Enforcer(\sigma' \cdot (\numTicks(\sigma_\sys'), D))
                        &\text{if } \PFun(\sigma_\sys') = D
                \end{cases}
            \end{align*}
            Similarly to the reactive case above, the final step follows
            from how the proactive update function $\PFun$ is defined
            by the enforcer update function $\EnfFun$ and
            how $\EnfFun$ is used by the enforcement algorithm to
            issue $\PCom$ and $\NCom$ commands.
        \end{itemize}
    \end{itemize}
    Finally, the finite case shows that there
    cannot exist an infinite system trace for which Lemma
    \ref{lem:enf-es-trace-correspondance}
    does not hold.
    For any infinite trace
    $\sigma_\sys \in \EStraces_\omega(\EventSystem(\Enforcer), \emptytrace_\sys)$
    we have shown that for all finite prefixes
    $\sigma_{\sys |\dots i}$
    with an arbitrary $i \in \Nat$,
    there exists an MFOTL
    trace $\sigma_f \in \Traces_f$ such that
    $\cConv(\sigma_{\sys |\dots i}) = \Enforcer(\sigma_f)$.
    If, for some infinite trace
    $\sigma_\sys \in \EStraces_\omega(\EventSystem(\Enforcer), \emptytrace_\sys)$,
    there would not exist a trace $\sigma_\omega \in \Traces_\omega$
    such that $\cConv(\sigma_\sys) = \Enforcer(\sigma_\omega)$,
    then there would exist some finite prefix
    $\sigma_{\sys |\dots i} \in \EStraces_f(\EventSystem(\Enforcer), \emptytrace_\sys)$
    such that there also does not exist a trace
    $\sigma_f \in \Traces_f$ such that
    $\cConv(\sigma_{\sys |\dots i}) = \Enforcer(\sigma_f)$.
\end{proof}
